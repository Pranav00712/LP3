def recursive_fibonacci(n):
    if n<=1:
        return n
    else:
        return recursive_fibonacci(n-1) + recursive_fibonacci(n-2)
        
def non_recursive_fibonacci(n):
    first=0
    second=1
    print(first)
    print(second)
    for i in range(2, n):
        third = first + second
        print(third)
        first = second
        second = third
